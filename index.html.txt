    function renderVerticalProductsList() {
      const container = document.getElementById("vertical-products-list");

      document.getElementById("vert-whatsapp-input").value = branchWhatsApp[branchKey] || "";

      let html = "";

      PRODUCTS.forEach(prod => {

    function switchTab(tabId) {
      document.querySelectorAll(".tab-panel").forEach(sec => sec.classList.add("hidden"));
      const target = document.getElementById(`tab-${tabId}`);
      if (target) target.classList.remove("hidden");

      document.querySelectorAll(".nav-tab").forEach(btn => {
        if (btn.getAttribute("data-tab") === tabId) {
          btn.className = "nav-tab px-3.5 py-2 rounded-lg text-xs sm:text-sm font-bold transition whitespace-nowrap bg-indigo-600 text-white shadow-sm flex items-center gap-1.5";
        } else {
          btn.className = "nav-tab px-3.5 py-2 rounded-lg text-xs sm:text-sm font-semibold transition whitespace-nowrap text-slate-300 hover:text-white hover:bg-slate-700/60 flex items-center gap-1.5";
        }
      });

      if (tabId === "invoice-studio") {
        renderInvoicePreview();
      } else if (tabId === "opex") {
        renderCustodyTable();
      } else if (tabId === "dashboard") {
        updateDashboard();
      }
    }

    function renderInvoicePreview() {
      const branchKey = document.getElementById("inv-select-branch").value;
      const phone = branchWhatsApp[branchKey] || "غير مسجل";
      document.getElementById("inv-whatsapp-display-input").value = phone;
      document.getElementById("inv-display-whatsapp").textContent = phone;
      document.getElementById("inv-display-client").textContent = branchKey.toUpperCase();
      document.getElementById("inv-display-approver").textContent = document.getElementById("inv-approved-by").value;

      const itemsData = orderingState[branchKey] || {};
      const tbody = document.getElementById("invoice-items-tbody");
      let rowsHtml = "";
      let totalQty = 0;
      let totalVal = 0;

      PRODUCTS.forEach(prod => {
        const q = itemsData[prod.id] || 0;
        if (q > 0) {
          const lineVal = q * prod.price;
          totalQty += q;
          totalVal += lineVal;
          rowsHtml += `
            <tr>
              <td class="py-2.5 px-3 border border-slate-200 text-center num-font">${prod.id}</td>
              <td class="py-2.5 px-3 border border-slate-200 font-extrabold text-slate-900">${prod.name}</td>
              <td class="py-2.5 px-3 border border-slate-200 text-center num-font font-bold text-indigo-700">${q}</td>
              <td class="py-2.5 px-3 border border-slate-200 text-center num-font">${prod.price} ج.م</td>
              <td class="py-2.5 px-3 border border-slate-200 text-center num-font font-extrabold">${lineVal.toLocaleString()} ج.م</td>
            </tr>
          `;
        }
      });

      }

      document.getElementById("inv-total-qty").textContent = `${totalQty} قطعة`;
      document.getElementById("inv-net-total").textContent = `${totalVal.toLocaleString()} ج.م`;
    }

    function sendWhatsAppInvoice() {
      const branchKey = document.getElementById("inv-select-branch").value;
      const phone = branchWhatsApp[branchKey];
      if (!phone) {
        alert("يرجى إدخال رقم واتساب صالح لهذا الفرع أولاً من شاشة الطلبيات!");
        return;
      }

      const approver = document.getElementById("inv-approved-by").value;
      const itemsData = orderingState[branchKey] || {};
      let text = `مرحباً بك من Urban Prep 🌿\nفاتورة توريد فرع: *${branchKey.toUpperCase()}*\n--------------------\n`;
      let total = 0;

      PRODUCTS.forEach(prod => {
        const q = itemsData[prod.id] || 0;
        if (q > 0) {
          const line = q * prod.price;
          total += line;
          text += `- ${prod.name}: (${q} قطعة) = ${line} ج.م\n`;
        }
      });

      text += `--------------------\n*الإجمالي: ${total} ج.م*\nمعتمدة بواسطة: ${approver} ✅`;
      const encoded = encodeURIComponent(text);
      window.open(`https://wa.me/${phone}?text=${encoded}`, '_blank');
    }

      const tbody = document.getElementById("custody-table-body");

      `).join("") || `<tr><td colspan="5" class="py-4 text-center text-slate-400">لا توجد عهد مسجلة بعد</td></tr>`;
    }

    function updateDashboard() {
      let sales = 0;
      Object.keys(orderingState).forEach(b => {
        const items = orderingState[b];
        PRODUCTS.forEach(p => { sales += (items[p.id] || 0) * p.price; });
      });
      let opex = custodyList.reduce((acc, x) => acc + x.amount, 0);
      document.getElementById("dash-sales").textContent = `${sales.toLocaleString()} ج.م`;
      document.getElementById("dash-opex").textContent = `${opex.toLocaleString()} ج.م`;
      document.getElementById("dash-profit").textContent = `${(sales - opex).toLocaleString()} ج.م`;
    }
