            { name: "م/ محمد النجار", username: "naggar", pass: "1234", role: "الإدارة", viewSales: true, canOpex: true, isAdmin: true },
            { name: "محمد أحمد", username: "mohamed", pass: "1111", role: "مدير مطبخ", viewSales: false, canOpex: true, isAdmin: false },
            { name: "مشرف تشغيل", username: "super", pass: "2222", role: "مشرف", viewSales: false, canOpex: false, isAdmin: false }
        ];

        let users = JSON.parse(localStorage.getItem('urbanPrepUsers')) || defaultUsers;
        let orders = JSON.parse(localStorage.getItem('urbanPrepOrders')) || [];
        let opexList = JSON.parse(localStorage.getItem('urbanPrepOpex')) || [];
        let currentUser = null;

        document.getElementById('orderDate').valueAsDate = new Date();
        document.getElementById('opexDate').valueAsDate = new Date();

        function handleLogin() {
            const u = document.getElementById('loginUser').value.trim().toLowerCase();
            const p = document.getElementById('loginPass').value.trim();
            const found = users.find(user => user.username.toLowerCase() === u && user.pass === p);

            if (!found) {
                document.getElementById('loginError').style.display = 'block';
                return;
            }
            currentUser = found;
            document.getElementById('loginError').style.display = 'none';
            document.getElementById('loginScreen').style.display = 'none';
            applyPermissions();
        }

        function handleLogout() {
            currentUser = null;
            document.getElementById('loginPass').value = '';
            document.getElementById('loginScreen').style.display = 'flex';
        }

        function applyPermissions() {
            document.getElementById('currentUserBadge').innerText = `${currentUser.name} - ${currentUser.role}`;
            document.getElementById('tabDashboard').classList.toggle('hidden', !currentUser.viewSales);
            document.getElementById('tabClients').classList.toggle('hidden', !currentUser.viewSales);
            document.getElementById('tabOpex').classList.toggle('hidden', !currentUser.canOpex);
            document.getElementById('tabUsers').classList.toggle('hidden', !currentUser.isAdmin);

            document.querySelectorAll('.sales-Perm-field, .sales-Perm-col').forEach(el => {
                el.classList.toggle('hidden', !currentUser.viewSales);
            });
            document.querySelectorAll('.delete-Perm-col').forEach(el => {
                el.classList.toggle('hidden', !currentUser.isAdmin);
            });

            if (!currentUser.isAdmin && currentUser.username === 'mohamed') {
                document.getElementById('giverSelect').value = "محمد أحمد";
            }

            if (currentUser.viewSales) switchTab('dashboard', document.getElementById('tabDashboard'));
            else switchTab('orders', document.getElementById('tabOrders'));
        }

        function switchTab(sectionId, btn) {
            document.querySelectorAll('.section').forEach(s => s.classList.remove('active'));
            document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
            document.getElementById(sectionId).classList.add('active');
            if (btn) btn.classList.add('active');
            renderAll();
        }

        function toggleOpexFields() {
            const t = document.getElementById('opexType').value;
            document.getElementById('receiverGroup').style.display = (t === 'تسليم عهدة') ? 'block' : 'none';
        }

        function syncProductPrice() {
            const sel = document.getElementById('productSelect');
            const price = sel.options[sel.selectedIndex].getAttribute('data-price');
            document.getElementById('unitPriceInput').value = price;
            calcOrderTotal();
        }

        function calcOrderTotal() {
            const q = parseFloat(document.getElementById('qtyInput').value) || 0;
            const up = parseFloat(document.getElementById('unitPriceInput').value) || 0;
            document.getElementById('priceInput').value = q * up;
        }

        function saveData() {
            localStorage.setItem('urbanPrepUsers', JSON.stringify(users));
            localStorage.setItem('urbanPrepOrders', JSON.stringify(orders));
            localStorage.setItem('urbanPrepOpex', JSON.stringify(opexList));
            renderAll();
        }

        function addOrder() {
            const date = document.getElementById('orderDate').value;
            const client = document.getElementById('clientSelect').value;
            const product = document.getElementById('productSelect').value;
            const qty = parseInt(document.getElementById('qtyInput').value) || 0;
            const sel = document.getElementById('productSelect');
            const defaultPrice = parseFloat(sel.options[sel.selectedIndex].getAttribute('data-price'));
            const unitPrice = currentUser.viewSales ? (parseFloat(document.getElementById('unitPriceInput').value) || defaultPrice) : defaultPrice;
            const price = qty * unitPrice;
            const status = currentUser.viewSales ? document.getElementById('paymentStatus').value : 'غير مدفوع';
            const whatsappPhone = document.getElementById('whatsappPhone').value.trim();

            if (qty > 0) {
                orders.push({ id: Date.now(), date, client, product, qty, price, status, whatsappPhone, addedBy: currentUser.name });
                saveData();
                alert('تم تسجيل الصنف للفرع بنجاح. يمكنك إضافة صنف آخر لنفس الفرع أو الانتقال لفرع جديد.');
            }
        }

        function addOpex() {
            const date = document.getElementById('opexDate').value;
            const type = document.getElementById('opexType').value;
            const giver = document.getElementById('giverSelect').value;
            const receiver = type === 'تسليم عهدة' ? document.getElementById('receiverSelect').value : '-';
            const desc = document.getElementById('descInput').value.trim();
            const amount = parseFloat(document.getElementById('amountInput').value) || 0;

            if (desc && amount > 0) {
                opexList.push({ id: Date.now(), date, type, giver, receiver, desc, amount, addedBy: currentUser.name });
                document.getElementById('descInput').value = '';
                document.getElementById('amountInput').value = '';
                saveData();
                alert('تم حفظ الحركة بنجاح');
            }
        }

        function deleteOrder(id) {
            if (!currentUser.isAdmin) return;
            if (confirm('تأكيد حذف الطلبية؟')) { orders = orders.filter(o => o.id !== id); saveData(); }
        }

        function deleteOpex(id) {
            if (!currentUser.isAdmin) return;
            if (confirm('تأكيد حذف المصروف أو العهدة؟')) { opexList = opexList.filter(o => o.id !== id); saveData(); }
        }

        function togglePaid(id) {
            if (!currentUser.viewSales) return;
            const ord = orders.find(o => o.id === id);
            if (ord) { ord.status = ord.status === 'مدفوع' ? 'غير مدفوع' : 'مدفوع'; saveData(); }
        }

        function sendWhatsApp(clientName) {
            const clientOrders = orders.filter(o => o.client === clientName);
            if (clientOrders.length === 0) { alert('لا توجد طلبات لهذا الفرع'); return; }
            const phone = clientOrders[clientOrders.length - 1].whatsappPhone || '';
            const total = clientOrders.reduce((s, o) => s + o.price, 0);
            
            let text = `مرحباً بك من Urban Prep 🌿\nفاتورة طلبات الفرع: *${clientName}*\n--------------------\n`;
            clientOrders.forEach(o => {
                text += `- ${o.product} (${o.qty} قطعة) = ${o.price} ج.م\n`;
            });
            text += `--------------------\n*الإجمالي الكلي: ${total} ج.م*\nتم الاعتماد بواسطة الإدارة ✅`;

            const encoded = encodeURIComponent(text);
            window.open(`https://wa.me/${phone}?text=${encoded}`, '_blank');
        }

        function saveNewUser() {
            if (!currentUser.isAdmin) return;
            const name = document.getElementById('newUserName').value.trim();
            const username = document.getElementById('newUserLogin').value.trim().toLowerCase();
            const pass = document.getElementById('newUserPass').value.trim();
            const role = document.getElementById('newUserRole').value;
            const viewSales = document.getElementById('permSales').value === 'yes';
            const canOpex = document.getElementById('permOpex').value === 'yes';
            const isAdmin = document.getElementById('permAdmin').value === 'yes';

            if (!name || !username || !pass) { alert('يرجى ملء جميع الحقول'); return; }
            const existingIndex = users.findIndex(u => u.username === username);
            const userObj = { name, username, pass, role, viewSales, canOpex, isAdmin };
            if (existingIndex >= 0) users[existingIndex] = userObj; else users.push(userObj);
            saveData(); alert('تم حفظ المستخدم والصلاحيات بنجاح');
        }

        function deleteUser(username) {
            if (username === 'gaber' || username === 'naggar') { alert('لا يمكن حذف حسابات الشركاء الأساسية!'); return; }
            if (confirm('حذف المستخدم؟')) { users = users.filter(u => u.username !== username); saveData(); }
        }

        function renderAll() {
            if (!currentUser) return;
            
            // خصم العهد المسلمة المستردة من إجمالي المصروفات الحسابية لو أمكن أو تجميعها
            const totalSales = orders.reduce((s, o) => s + o.price, 0);
            const totalOpex = opexList.reduce((s, i) => s + i.amount, 0);
            const netProfit = totalSales - totalOpex;
            const share = (netProfit / 2).toFixed(2);

            document.getElementById('dashSales').innerText = totalSales + ' ج.م';
            document.getElementById('dashOpex').innerText = totalOpex + ' ج.م';
            document.getElementById('dashProfit').innerText = netProfit + ' ج.م';
            document.getElementById('dashNaggar').innerText = share + ' ج.م';
            document.getElementById('dashGaber').innerText = share + ' ج.م';

            const oBody = document.getElementById('ordersTableBody');
            oBody.innerHTML = '';
            orders.forEach(o => {
                const tr = document.createElement('tr');
                tr.innerHTML = `<td>${o.date}</td><td><strong>${o.client}</strong></td><td>${o.product}</td><td>${o.qty}</td>
                    <td class="${currentUser.viewSales ? '' : 'hidden'}">${o.price} ج.م</td>
                    <td class="${currentUser.viewSales ? '' : 'hidden'}"><span class="status-${o.status === 'مدفوع' ? 'paid' : 'unpaid'}" style="cursor:pointer" onclick="togglePaid(${o.id})">${o.status}</span></td>
                    <td>${o.whatsappPhone || '-'}</td>
                    <td><small>${o.addedBy}</small></td>
                    <td class="${currentUser.isAdmin ? '' : 'hidden'}"><button class="delete-btn" onclick="deleteOrder(${o.id})">حذف</button></td>`;
                oBody.appendChild(tr);
            });

            const opBody = document.getElementById('opexTableBody');
            opBody.innerHTML = '';
            opexList.forEach(i => {
                const tr = document.createElement('tr');
                tr.innerHTML = `<td>${i.date}</td><td><strong>${i.type}</strong></td><td>${i.giver}</td><td>${i.receiver}</td><td>${i.desc}</td><td>${i.amount} ج.م</td>
                    <td><small>${i.addedBy || '-'}</small></td>
                    <td class="${currentUser.isAdmin ? '' : 'hidden'}"><button class="delete-btn" onclick="deleteOpex(${i.id})">حذف</button></td>`;
                opBody.appendChild(tr);
            });

            if (currentUser.viewSales) {
                const cBody = document.getElementById('clientsTableBody');
                cBody.innerHTML = '';
                [...new Set(orders.map(o => o.client))].forEach(cn => {
                    const cOrders = orders.filter(o => o.client === cn);
                    const tQty = cOrders.reduce((s, o) => s + o.qty, 0);
                    const tVal = cOrders.reduce((s, o) => s + o.price, 0);
                    const paid = cOrders.filter(o => o.status === 'مدفوع').reduce((s, o) => s + o.price, 0);
                    const tr = document.createElement('tr');
                    tr.innerHTML = `<td><strong>${cn}</strong></td><td>${tQty}</td><td>${tVal} ج.م</td><td class="status-paid">${paid} ج.م</td><td class="status-unpaid">${tVal - paid} ج.م</td>
                        <td><button class="whatsapp-btn" onclick="sendWhatsApp('${cn}')">إرسال واتس 📱</button></td>`;
                    cBody.appendChild(tr);
                });
            }

            if (currentUser.isAdmin) {
                const uBody = document.getElementById('usersTableBody');
                uBody.innerHTML = '';
                users.forEach(u => {
                    const tr = document.createElement('tr');
                    tr.innerHTML = `<td><strong>${u.name}</strong></td><td>${u.username}</td><td>${u.pass}</td><td>${u.role}</td><td>${u.viewSales ? '✅' : '❌'}</td><td>${u.canOpex ? '✅' : '❌'}</td><td>${u.isAdmin ? '✅' : '❌'}</td>
                        <td>${(u.username !== 'gaber' && u.username !== 'naggar') ? `<button class="delete-btn" onclick="deleteUser('${u.username}')">حذف</button>` : 'أساسي'}</td>`;
                    uBody.appendChild(tr);
                });
            }
        }
    </script>
